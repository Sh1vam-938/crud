  YOUR COMPLETE API
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  AUTH (no token needed):
    POST   /api/auth/register   → create account (rate: 10/15min)
    POST   /api/auth/login      → get JWT token  (rate: 10/15min)

  TASKS (JWT token required):
    GET    /api/tasks           → all YOUR tasks (filter by status/priority)
    GET    /api/tasks/stats     → counts by status and priority
    GET    /api/tasks/:id       → one task by id
    POST   /api/tasks           → create task
    PUT    /api/tasks/:id       → update task fields
    PATCH  /api/tasks/:id/status→ quick status update
    DELETE /api/tasks/:id       → delete task

  YOUR SECURITY STACK
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  helmet         → 11+ secure HTTP headers (XSS, clickjacking, etc.)
  cors           → controls who can call your API
  rate-limit     → blocks brute force and DDoS attacks
  JWT            → stateless auth tokens with expiry
  bcrypt         → passwords hashed — safe even if DB is stolen
  parameterized  → SQL injection impossible
  express-validator → bad data rejected before DB
  user scoping   → WHERE user_id = ? everywhere (IDOR prevention)
  error handler  → stack traces hidden in production
